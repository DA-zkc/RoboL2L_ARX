cmake_minimum_required(VERSION 3.19)
project(franka_cap LANGUAGES CXX)
find_package(Franka 0.19 REQUIRED)
add_executable(franka_cap_state state.cpp)
target_compile_features(franka_cap_state PRIVATE cxx_std_17)
target_compile_options(franka_cap_state PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(franka_cap_state PRIVATE Franka::Franka)
set(ORBBEC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/orbbec" CACHE PATH "Orbbec SDK installation")
add_executable(franka_cap_global global_camera.cpp)
target_compile_features(franka_cap_global PRIVATE cxx_std_17)
target_include_directories(franka_cap_global PRIVATE "${ORBBEC_ROOT}/include")
target_link_directories(franka_cap_global PRIVATE "${ORBBEC_ROOT}/lib")
target_link_libraries(franka_cap_global PRIVATE OrbbecSDK pthread)
set_target_properties(franka_cap_global PROPERTIES BUILD_RPATH "${ORBBEC_ROOT}/lib")

add_executable(franka_cap_motion motion.cpp)
target_compile_features(franka_cap_motion PRIVATE cxx_std_17)
target_compile_options(franka_cap_motion PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(franka_cap_motion PRIVATE Franka::Franka)
add_executable(franka_cap_gripper gripper.cpp)
target_compile_features(franka_cap_gripper PRIVATE cxx_std_17)
target_compile_options(franka_cap_gripper PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(franka_cap_gripper PRIVATE Franka::Franka pthread)
